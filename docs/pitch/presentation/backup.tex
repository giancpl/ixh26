\begin{frame}[noframenumbering]{The verifier cannot distinguish identical capture transcripts}
\begin{pitchcard}
\textbf{Model}\par
An adversary controls OS + app and can invoke an enrolled signing key.\par
The verifier sees only $(J,\mathit{receipt},\mathit{bundle})$.
\end{pitchcard}
\vspace{0.3cm}
\begin{align*}
\mathit{physical\ capture\ of}\ J&\longrightarrow\mathit{transcript}\ \tau\\
\mathit{injection\ of\ the\ same}\ J&\longrightarrow\mathit{transcript}\ \tau
\end{align*}
\vspace{0.2cm}
Without an independently authenticated capture observation, the same transcript
does not identify which history occurred.
\vfill
\labelsmall{Model-level argument, not a universal impossibility theorem.\par
This does not let the client forge a vendor credential or pass every independent check.}
\end{frame}

\begin{frame}[noframenumbering]{Missing evidence never upgrades an origin level}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{8.6cm}>{\raggedright\arraybackslash}p{4.1cm}@{}}
\textbf{Condition, in priority order} & \textbf{Result}\\\hline
Definitive failure / conflict / invalid mandatory check & Failed\\
Mandatory check unresolved & Pending\\
Manufacturer passes trust + asset binding & Manufacturer level\\
Manufacturer not decisive; trusted, calibrated PRNU pass & PRNU level\\
Neither passes or fails; mandatory checks valid & No origin verification\\
\end{tabular}\par
\vspace{0.3cm}
\labelsmall{Codes: 0 unavailable / 1 inconclusive / 2 pass / 3 fail.\par
Both checks are assessed. A calibrated PRNU mismatch blocks even a manufacturer pass.\par
An uncalibrated low score is not a mismatch. Personal preference never changes the level or $C$.}
\end{frame}

\begin{frame}[noframenumbering]{PRNU supports statistical compatibility, not sensor proof}
\begin{align*}
W_i &= I_i-\mathrm{denoise}(I_i), &
K[p] &= \frac{\sum_i m_i[p]I_i[p]W_i[p]}{\sum_i m_i[p]I_i[p]^2}\\[0.35cm]
\mathsf{SPCE} &= \frac{\operatorname{sign}(r_0)r_0^2}
{\operatorname{mean}_{s\notin\mathcal E}\,r(s)^2}

$\mathcal E$: pinned central exclusion window. Invalid area/zero energy gives inconclusive.}

\textbf{Match / mismatch:} independently calibrated decision regions.\par
\textbf{Otherwise:} inconclusive / unavailable.\par

\begin{frame}[noframenumbering]{Hybrid signatures do not make the whole trust chain post-quantum}
\vspace{0.3cm}
{\Large\(\mathsf{ML\text{-}DSA\text{-}65}\quad\mathbf{AND}\quad\mathsf{Ed25519}\)}\par
\vspace{0.4cm}
\begin{pitchcard}
Both signatures must verify; separate wire version 2.\par
Software keys wrapped at rest by hardware Keystore AES-GCM.\par

Hybrid secrets enter RAM during signing; not hardware-native.
\end{pitchcard}
\vfill

\end{frame}
